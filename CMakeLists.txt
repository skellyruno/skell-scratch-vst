cmake_minimum_required(VERSION 3.22)
project(ScratchPad VERSION 1.7.0 LANGUAGES C CXX)
include(FetchContent)
if(DEFINED JUCE_PATH AND EXISTS "${JUCE_PATH}/CMakeLists.txt")
  add_subdirectory("${JUCE_PATH}" JUCE)
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(ScratchPad COMPANY_NAME "Skell Audio" BUNDLE_ID "audio.skell.scratchpad" PLUGIN_MANUFACTURER_CODE SkAu PLUGIN_CODE ScP1 FORMATS VST3 Standalone PRODUCT_NAME "ScratchPad" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(ScratchPadAssets SOURCES Assets/TechFont.otf)
juce_generate_juce_header(ScratchPad)
target_sources(ScratchPad PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/CurveMath.h Source/HeadSafety.h Source/PluginEditor.cpp Source/PluginEditor.h Source/ui/Theme.h Source/ui/StopButton.h Source/ui/CurveEditor.h Source/ui/CurveEditor.cpp Source/ui/ScratchKnob.h Source/ui/ScratchKnob.cpp Source/ui/FooterBar.h Source/ui/FooterBar.cpp)
target_compile_features(ScratchPad PRIVATE cxx_std_17)
target_compile_definitions(ScratchPad PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ScratchPad PRIVATE ScratchPadAssets juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra juce::juce_dsp)

# Offline core tests (no JUCE linkage required); run with `ctest --test-dir build -C Release`.
include(CTest)
if(BUILD_TESTING)
  add_executable(ScratchPadCurveTests tests/curve_math_test.cpp)
  add_executable(ScratchPadHeadSafetyTests tests/head_safety_test.cpp)
  add_executable(ScratchPadPresetTests tests/preset_bank_test.cpp)
  target_compile_features(ScratchPadCurveTests PRIVATE cxx_std_17)
  target_compile_features(ScratchPadHeadSafetyTests PRIVATE cxx_std_17)
  target_compile_features(ScratchPadPresetTests PRIVATE cxx_std_17)
  add_test(NAME CurveMath COMMAND ScratchPadCurveTests)
  add_test(NAME HeadSafety COMMAND ScratchPadHeadSafetyTests)
  add_test(NAME PresetBank COMMAND ScratchPadPresetTests)
endif()
